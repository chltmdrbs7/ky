<script>
(function() {
    'use strict';

    // 네임스페이스 생성으로 충돌 방지
    if (!window.PrimeRoadHeaderManager) {
        window.PrimeRoadHeaderManager = {};
    }

    // 이전 인스턴스 정리
    if (window.PrimeRoadHeaderManager.cleanup) {
        window.PrimeRoadHeaderManager.cleanup();
    }

    // 초기화 함수
    function initializePrimeRoadHeader() {
        const section = document.getElementById('s2025101755ff003c1e853');
        if (!section) {
            console.warn('PrimeRoad Header: Section not found');
            return false;
        }

        const hamburger = section.querySelector('#gnHamburger');
        const fullscreenMenu = section.querySelector('#gnMenu');
        const menuLinks = section.querySelectorAll('.gn-menu-link');
        const header = section.querySelector('.gn-header-wrapper');

        if (!hamburger || !fullscreenMenu) {
            console.warn('PrimeRoad Header: Required elements not found');
            return false;
        }

        // 메뉴 상태 관리
        let isMenuOpen = false;

        // 메뉴 토글 함수
        function toggleMenu(forceClose = false) {
            if (forceClose || isMenuOpen) {
                // 메뉴 닫기
                hamburger.classList.remove('active');
                fullscreenMenu.classList.remove('active');
                document.body.style.overflow = '';
                isMenuOpen = false;
            } else {
                // 메뉴 열기
                hamburger.classList.add('active');
                fullscreenMenu.classList.add('active');
                document.body.style.overflow = 'hidden';
                isMenuOpen = true;
            }
        }

        // 이벤트 핸들러 함수들
        function handleHamburgerClick(e) {
            e.preventDefault();
            e.stopPropagation();
            toggleMenu();
        }

        function handleMenuLinkClick() {
            setTimeout(() => toggleMenu(true), 100);
        }

        function handleOutsideClick(e) {
            if (e.target === fullscreenMenu && isMenuOpen) {
                toggleMenu(true);
            }
        }

        function handleEscKey(e) {
            if (e.key === 'Escape' && isMenuOpen) {
                toggleMenu(true);
            }
        }

        function handleScroll() {
            if (header) {
                if (window.scrollY > 50) {
                    header.classList.add('scrolled');
                } else {
                    header.classList.remove('scrolled');
                }
            }
        }

        // 이벤트 리스너 등록
        hamburger.addEventListener('click', handleHamburgerClick);

        menuLinks.forEach(link => {
            link.addEventListener('click', handleMenuLinkClick);
        });

        fullscreenMenu.addEventListener('click', handleOutsideClick);
        document.addEventListener('keydown', handleEscKey);
        window.addEventListener('scroll', handleScroll);

        // 초기 스크롤 상태 확인
        handleScroll();

        // cleanup 함수 정의
        window.PrimeRoadHeaderManager.cleanup = function() {
            try {
                hamburger.removeEventListener('click', handleHamburgerClick);
                menuLinks.forEach(link => {
                    link.removeEventListener('click', handleMenuLinkClick);
                });
                fullscreenMenu.removeEventListener('click', handleOutsideClick);
                document.removeEventListener('keydown', handleEscKey);
                window.removeEventListener('scroll', handleScroll);

                // 메뉴가 열려있으면 닫기
                if (isMenuOpen) {
                    toggleMenu(true);
                }
            } catch (e) {
                console.warn('PrimeRoad Header cleanup error:', e);
            }
        };

        console.log('PrimeRoad Header initialized successfully');
        return true;
    }

    // 초기화 시도 함수
    function tryInitialize() {
        const success = initializePrimeRoadHeader();
        if (success) {
            window.PrimeRoadHeaderManager.initialized = true;
        }
        return success;
    }

    // DOM 준비 상태에 따른 초기화
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', tryInitialize);
    } else {
        tryInitialize();
    }

    // 아임웹 환경 대응 - 지연 초기화
    setTimeout(tryInitialize, 100);
    setTimeout(tryInitialize, 500);

    // MutationObserver로 DOM 변경 감지
    if (!window.PrimeRoadHeaderManager.observer) {
        window.PrimeRoadHeaderManager.observer = new MutationObserver(function(mutations) {
            const headerExists = document.getElementById('gnHamburger');
            if (headerExists && !window.PrimeRoadHeaderManager.initialized) {
                tryInitialize();
            }
        });

        // body 관찰 시작
        if (document.body) {
            window.PrimeRoadHeaderManager.observer.observe(document.body, {
                childList: true,
                subtree: true
            });
        }
    }

})();
</script>
