// CTA 버튼 및 카드 마우스 추적 효과
(function() {
    // 데스크톱에서만 실행
    if (window.innerWidth > 768) {
        const ctaButtons = document.querySelectorAll('.ha-cta-btn');

        ctaButtons.forEach(button => {
            button.addEventListener('mousemove', function(e) {
                const rect = this.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;

                this.style.setProperty('--mx', `${x}px`);
                this.style.setProperty('--my', `${y}px`);
            });

            button.addEventListener('mouseleave', function() {
                this.style.setProperty('--mx', '50%');
                this.style.setProperty('--my', '50%');
            });
        });

        // 카드들에도 마우스 추적 스포트라이트 효과 추가
        const cards = document.querySelectorAll('.ha-rate-card, .ha-info-item');

        cards.forEach(card => {
            card.addEventListener('mousemove', function(e) {
                const rect = this.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;

                // 스포트라이트 효과용 마우스 위치 업데이트
                this.style.setProperty('--mx', `${x}px`);
                this.style.setProperty('--my', `${y}px`);

                // 3D 틸트 효과
                const centerX = rect.width / 2;
                const centerY = rect.height / 2;

                const rotateX = (y - centerY) / 20;
                const rotateY = (centerX - x) / 20;

                this.style.transform = `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) translateY(-3px)`;
            });

            card.addEventListener('mouseleave', function() {
                this.style.transform = '';
                this.style.setProperty('--mx', '50%');
                this.style.setProperty('--my', '50%');
            });
        });
    }
})();
</script>
