function showTab(tabName) {
    const tabContents = document.querySelectorAll('#s202510174849d7fcac602 .tab-content');
    tabContents.forEach(content => {
        content.classList.remove('active');
    });

    const tabButtons = document.querySelectorAll('#s202510174849d7fcac602 .tab-button');
    tabButtons.forEach(button => {
        button.classList.remove('active');
    });

    document.getElementById(tabName).classList.add('active');
    event.target.closest('.tab-button').classList.add('active');

    if (window.innerWidth <= 768) {
        const tabContent = document.getElementById(tabName);
        if (tabContent) {
            setTimeout(() => {
                tabContent.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            }, 100);
        }
    }
}

function focusCompanyInput(event) {
    event.preventDefault();
    const targetSection = document.querySelector('#s2025101713ec37ac558db');
    if (targetSection) {
        targetSection.scrollIntoView({
            behavior: 'smooth',
            block: 'start'
        });

        setTimeout(() => {
            const companyInput = targetSection.querySelector('input[name="company"]') ||
                                targetSection.querySelector('input[placeholder*="회사명"]') ||
                                targetSection.querySelector('input[type="text"]:first-of-type');

            if (companyInput) {
                companyInput.focus();
                companyInput.click();

                setTimeout(() => {
                    companyInput.scrollIntoView({
                        behavior: 'smooth',
                        block: 'center'
                    });
                }, 300);
            }
        }, 800);
    }
}

// 스포트라이트 마우스 트래킹 - 작은 카드에만 적용
document.addEventListener('DOMContentLoaded', function() {
    const sectionElement = document.querySelector('#s202510174849d7fcac602');
    if (!sectionElement) return;

    // info-item과 benefit-card에만 스포트라이트 적용
    const smallCards = sectionElement.querySelectorAll('.info-item, .benefit-card');

    smallCards.forEach(card => {
        card.addEventListener('mousemove', function(e) {
            const rect = this.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            this.style.setProperty('--mx', x + 'px');
            this.style.setProperty('--my', y + 'px');
        });

        card.addEventListener('mouseleave', function() {
            this.style.removeProperty('--mx');
            this.style.removeProperty('--my');
        });
    });
});
</script>
