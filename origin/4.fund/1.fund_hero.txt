 function scrollToForm(event) {
   event.preventDefault();
   const formSection = document.querySelector('#s202510177c8482d36086d');
   if (formSection) {
     const companyInput = formSection.querySelector('input[name="company"]');
     if (companyInput) {
       companyInput.scrollIntoView({ behavior: 'smooth', block: 'center' });
       setTimeout(() => { companyInput.focus(); }, 800);
     } else {
       formSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
     }
   }
 }

         const rect = this.getBoundingClientRect();

         const y = e.clientY - rect.top;

         this.style.setProperty('--mx', x + 'px');
         this.style.setProperty('--my', y + 'px');

       });

       btn.addEventListener('mouseleave', function() {

     });

 });
