  <script>
  document.addEventListener('DOMContentLoaded', function() {
      const sectionElement = document.querySelector('#s202510178a1cb6811ca81');
      if (!sectionElement) return;

      // 스포트라이트 마우스 트래킹 - 전문가 카드
      const expertCards = sectionElement.querySelectorAll('.expert-card');

      expertCards.forEach(card => {
          card.addEventListener('mousemove', function(e) {
              const rect = this.getBoundingClientRect();
              const x = e.clientX - rect.left;
              const y = e.clientY - rect.top;
              this.style.setProperty('--mx', x + 'px');
              this.style.setProperty('--my', y + 'px');
          });

          card.addEventListener('mouseleave', function() {
              this.style.removeProperty('--mx');
              this.style.removeProperty('--my');
          });
      });

      // 모바일 가로 스크롤 네비게이션
      const expertWrapper = sectionElement.querySelector('.expert-grid-wrapper');
      const expertGrid = expertWrapper ? expertWrapper.querySelector('.expert-grid') : null;
      const expertPrev = expertWrapper ? expertWrapper.querySelector('.scroll-nav.prev') : null;
      const expertNext = expertWrapper ? expertWrapper.querySelector('.scroll-nav.next') : null;

      if (expertGrid && expertPrev && expertNext) {
          const cards = expertGrid.querySelectorAll('.expert-card');

          function checkExpertScroll() {
              const scrollLeft = expertGrid.scrollLeft;
              const maxScroll = expertGrid.scrollWidth - expertGrid.clientWidth;

              if (scrollLeft <= 10) {
                  expertPrev.classList.add('hidden');
              } else {
                  expertPrev.classList.remove('hidden');
              }

              if (scrollLeft >= maxScroll - 10) {
                  expertNext.classList.add('hidden');
              } else {
                  expertNext.classList.remove('hidden');
              }
          }

          // 초기 체크 및 리사이즈 이벤트
          function initScrollNav() {
              if (window.innerWidth <= 768) {
                  checkExpertScroll();
              }
          }

          initScrollNav();
          window.addEventListener('resize', initScrollNav);
          expertGrid.addEventListener('scroll', checkExpertScroll);

          expertPrev.addEventListener('click', function(e) {
              e.preventDefault();
              e.stopPropagation();
              const cardWidth = cards[0].offsetWidth + 16;
              expertGrid.scrollBy({
                  left: -cardWidth,
                  behavior: 'smooth'
              });
          });

          expertNext.addEventListener('click', function(e) {
              e.preventDefault();
              e.stopPropagation();
              const cardWidth = cards[0].offsetWidth + 16;
              expertGrid.scrollBy({
                  left: cardWidth,
                  behavior: 'smooth'
              });
          });
      }
  });
  </script>
