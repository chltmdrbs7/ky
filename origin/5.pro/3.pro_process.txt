  // 스포트라이트 마우스 트래킹
  document.addEventListener('DOMContentLoaded', function() {
      const sectionElement = document.querySelector('#s2025101722cf7e1a99bd1');
      if (!sectionElement) return;

      // 스텝 콘텐츠 스포트라이트
      const stepContents = sectionElement.querySelectorAll('.step-content');
      stepContents.forEach(content => {
          content.addEventListener('mousemove', function(e) {
              const rect = this.getBoundingClientRect();
              const x = e.clientX - rect.left;
              const y = e.clientY - rect.top;
              this.style.setProperty('--mx', x + 'px');
              this.style.setProperty('--my', y + 'px');
          });

          content.addEventListener('mouseleave', function() {
              this.style.removeProperty('--mx');
              this.style.removeProperty('--my');
          });
      });

      // CTA 버튼 스포트라이트
      const ctaButtons = sectionElement.querySelectorAll('.cta-button-primary');
      ctaButtons.forEach(button => {
          button.addEventListener('mousemove', function(e) {
              const rect = this.getBoundingClientRect();
              const x = e.clientX - rect.left;
              const y = e.clientY - rect.top;
              this.style.setProperty('--mx', x + 'px');
              this.style.setProperty('--my', y + 'px');
          });

          button.addEventListener('mouseleave', function() {
              this.style.removeProperty('--mx');
              this.style.removeProperty('--my');
          });
      });

      // 모바일 탭 + 스와이프 기능
      const tabs = sectionElement.querySelectorAll('.process-tab');
      const wrapper = sectionElement.querySelector('.swipe-wrapper');
      const steps = sectionElement.querySelectorAll('.process-step');

      if (!wrapper || steps.length === 0) return;

      let currentStep = 0;
      let startX = 0;
      let isDragging = false;

      // 탭 클릭 이벤트
      tabs.forEach(tab => {
          tab.addEventListener('click', function() {
              const stepIndex = parseInt(this.getAttribute('data-step'));
              goToStep(stepIndex);
          });
      });

      // 스텝 이동 함수
      function goToStep(index) {
          if (index < 0 || index >= steps.length) return;

          currentStep = index;

          // 모바일에서만 active 클래스 적용
          if (window.innerWidth <= 768) {
              // 모든 스텝 숨김
              steps.forEach(step => step.classList.remove('active'));
              // 선택된 스텝만 표시
              steps[index].classList.add('active');
          }

          // 탭 활성화
          tabs.forEach(tab => tab.classList.remove('active'));
          tabs[index].classList.add('active');
      }

      // 터치 스와이프 이벤트
      if (window.innerWidth <= 768) {
          wrapper.addEventListener('touchstart', function(e) {
              startX = e.touches[0].clientX;
              isDragging = true;
          });

          wrapper.addEventListener('touchmove', function(e) {
              if (!isDragging) return;
              e.preventDefault();
          }, { passive: false });

          wrapper.addEventListener('touchend', function(e) {
              if (!isDragging) return;
              isDragging = false;

              const endX = e.changedTouches[0].clientX;
              const diff = startX - endX;

              // 50px 이상 스와이프 시 페이지 전환
              if (Math.abs(diff) > 50) {
                  if (diff > 0 && currentStep < steps.length - 1) {
                      // 왼쪽 스와이프 - 다음
                      goToStep(currentStep + 1);
                  } else if (diff < 0 && currentStep > 0) {
                      // 오른쪽 스와이프 - 이전
                      goToStep(currentStep - 1);
                  }
              }
          });
      }

      // 화면 크기 변경 시 처리
      let resizeTimer;
      window.addEventListener('resize', function() {
          clearTimeout(resizeTimer);
          resizeTimer = setTimeout(function() {
              if (window.innerWidth > 768) {
                  // PC에서는 모든 스텝 표시
                  steps.forEach(step => step.classList.remove('active'));
              } else {
                  // 모바일에서는 현재 스텝만 표시
                  goToStep(currentStep);
              }
          }, 100);
      });

      // 초기 로드 시 위치 설정
      if (window.innerWidth <= 768) {
          goToStep(0);
      }
  });
  </script>
