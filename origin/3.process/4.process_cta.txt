<script>
function scrollToForm(event) {
  event.preventDefault();

  // 입력폼 섹션을 찾음
  const formSection = document.querySelector('#s20251017b3d28f5e4a1c7');

  if (formSection) {
      // 입력폼 섹션 내의 회사명 입력 필드를 찾음
      const companyInput = formSection.querySelector('input[name="company"]');

      if (companyInput) {
          // 회사명 입력 필드로 스크롤하고 포커스
          companyInput.scrollIntoView({
              behavior: 'smooth',
              block: 'center'
          });

          // 약간의 지연 후 포커스 (스크롤 완료 대기)
          setTimeout(() => {
              companyInput.focus();
          }, 800);
      } else {
          // 회사명 입력 필드가 없으면 섹션으로 스크롤
          formSection.scrollIntoView({
              behavior: 'smooth',
              block: 'start'
          });
      }
  }
}
</script>
